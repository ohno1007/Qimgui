cmake_minimum_required(VERSION 3.18)
project(MD3Consumer LANGUAGES CXX)
option(MD3_CONSUMER_INSTALLED "Use an installed QimguiMD3 package" OFF)
option(MD3_CONSUMER_SHOWCASE "Also exercise the optional gallery library" OFF)

if(MD3_CONSUMER_INSTALLED)
    if(MD3_CONSUMER_SHOWCASE)
        find_package(QimguiMD3 0.1 REQUIRED CONFIG COMPONENTS MD3 Showcase)
    else()
        find_package(QimguiMD3 0.1 REQUIRED CONFIG COMPONENTS MD3)
    endif()
else()
    get_filename_component(_qimgui_default "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
    set(QIMGUI_SOURCE_DIR "${_qimgui_default}" CACHE PATH "Qimgui checkout")
    set(QIMGUI_MD3_BUILD_SHOWCASE ${MD3_CONSUMER_SHOWCASE})
    set(QIMGUI_MD3_INSTALL OFF)
    add_subdirectory("${QIMGUI_SOURCE_DIR}/libraries/md3" qimgui-md3)
endif()

add_executable(md3_consumer main.cpp)
target_link_libraries(md3_consumer PRIVATE Qimgui::MD3)
if(MD3_CONSUMER_SHOWCASE)
    target_compile_definitions(md3_consumer PRIVATE MD3_CONSUMER_SHOWCASE=1)
    target_link_libraries(md3_consumer PRIVATE Qimgui::MD3Showcase)
endif()
